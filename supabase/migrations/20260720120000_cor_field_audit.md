-- Enums
DO $$ BEGIN
  CREATE TYPE "CorSampleProjectType" AS ENUM (
    'TOWER_CRANE','MOBILE_CRANE','CIVIL','INDUSTRIAL','COMMERCIAL','RESIDENTIAL',
    'FABRICATION_SHOP','YARD','MAINTENANCE','WAREHOUSE','OFFICE'
  );
EXCEPTION WHEN duplicate_object THEN NULL; END $$;

DO $$ BEGIN
  CREATE TYPE "CorSampleRiskLevel" AS ENUM ('LOW','MEDIUM','HIGH','CRITICAL');
EXCEPTION WHEN duplicate_object THEN NULL; END $$;

DO $$ BEGIN
  CREATE TYPE "CorSampleSiteStatus" AS ENUM ('PLANNED','ACTIVE','COMPLETE','ON_HOLD','CANCELLED');
EXCEPTION WHEN duplicate_object THEN NULL; END $$;

DO $$ BEGIN
  CREATE TYPE "CorInterviewType" AS ENUM ('WORKER','MANAGER');
EXCEPTION WHEN duplicate_object THEN NULL; END $$;

DO $$ BEGIN
  CREATE TYPE "CorInterviewStatus" AS ENUM ('SCHEDULED','IN_PROGRESS','COMPLETED','SKIPPED','INCOMPLETE');
EXCEPTION WHEN duplicate_object THEN NULL; END $$;

DO $$ BEGIN
  CREATE TYPE "CorInterviewAnswer" AS ENUM ('PASS','FAIL','NA','NOT_ASKED');
EXCEPTION WHEN duplicate_object THEN NULL; END $$;

DO $$ BEGIN
  CREATE TYPE "CorObservationCategory" AS ENUM (
    'HOUSEKEEPING','PPE','RIGGING','EQUIPMENT','MOBILE_EQUIPMENT','TOWER_CRANE',
    'FALL_PROTECTION','TRAFFIC_CONTROL','EXCAVATION','CONFINED_SPACE','ELECTRICAL',
    'LADDERS','SCAFFOLDING','MATERIAL_STORAGE','ENVIRONMENTAL'
  );
EXCEPTION WHEN duplicate_object THEN NULL; END $$;

DO $$ BEGIN
  CREATE TYPE "CorObservationResult" AS ENUM ('PASS','FAIL','NA');
EXCEPTION WHEN duplicate_object THEN NULL; END $$;
